-- =====================================================
-- VERIFICATION QUERY
-- =====================================================
-- Run this to verify the data was inserted correctly:
SELECT 
    cq.id as main_q_id,
    cq.question_order,
    LEFT(cq.question_text, 60) as question_preview,
    COUNT(DISTINCT csq.id) as sub_questions,
    COUNT(cso.id) as total_options
FROM composite_questions cq
LEFT JOIN composite_sub_questions csq ON csq.composite_question_id = cq.id
LEFT JOIN composite_sub_options cso ON cso.composite_sub_question_id = csq.id
WHERE cq.composite_exam_id = 1
GROUP BY cq.id
ORDER BY cq.question_order;
